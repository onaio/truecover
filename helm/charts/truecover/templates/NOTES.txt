Get the application URLs by running these commands:

Frontend:
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "truecover.name" . }},app.kubernetes.io/instance={{ .Release.Name }},app.kubernetes.io/component=frontend" -o jsonpath="{.items[0].metadata.name}")
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:80

Backend API:
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "truecover.name" . }},app.kubernetes.io/instance={{ .Release.Name }},app.kubernetes.io/component=backend" -o jsonpath="{.items[0].metadata.name}")
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 5000:5000

  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 3000:3000

NOTE: The frontend image must be built with the required environment variables:
  - VITE_MAPBOX_TOKEN
  - VITE_CLERK_PUBLISHABLE_KEY
  - VITE_API_URL
  - VITE_MARTIN_URL

These are baked into the image at build time using Docker build args.
